class Solution:
    def searchInsert(self, nums: list[int], target: int) -> int:

        left = 0                    # Start of the search area
        right = len(nums) - 1       # End of the search area

        while left <= right:       
            mid = (left + right) // 2 

            if nums[mid] == target:     
                return mid              

            elif nums[mid] < target:    
                left = mid + 1           

            else:                       
                right = mid - 1         

        return left
