class Solution:
    def searchInsert(self, nums: list[int], target: int) -> int:
        left = 0
        # left starts at the first index
        right = len(nums) - 1
        # right starts at the last index
        while left <= right:
            mid = (left + right) // 2
            if nums[mid] == target:
                return mid
            elif nums[mid] < target:
                left = mid + 1
            else:
                right = mid - 1
        return left
