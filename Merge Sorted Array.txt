class Solution:
    def merge(self, nums1: list[int], m: int, nums2: list[int], n: int) -> None:

        # i points to the last real element in nums1
        i = m - 1

        # j points to the last element in nums2
        j = n - 1

        # k points to the last position in nums1
        k = m + n - 1

        # Continue while nums2 still has elements
        while j >= 0:

            # If nums1 still has elements
            # and its current value is bigger
            if i >= 0 and nums1[i] > nums2[j]:

                # Put nums1's value at the back
                nums1[k] = nums1[i]

                # Move nums1 pointer backward
                i -= 1

            else:

                # Put nums2's value at the back
                nums1[k] = nums2[j]

                # Move nums2 pointer backward
                j -= 1

            # Move the position where we write backward
            k -= 1
